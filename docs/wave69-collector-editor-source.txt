    def _collector_editor_dialog(self, title="Collector", initial_name="", initial_areas=None, initial_notes=""):
        """Unified editor for Collector name + route areas + notes."""
        from tkinter import messagebox

        top = tk.Toplevel(self.root)
        top.title(title)
        top.transient(self.root)
        try:
            top.grab_set()
        except Exception as __spina_exc:
            _log_suppressed_once('excpass_0236', 'suppressed exception excpass_0236', __spina_exc)
            pass
        # Size/position: fit within screen so OK/Cancel never fall off-screen (high-DPI safe)
        try:
            top.update_idletasks()
            sw = top.winfo_screenwidth()
            sh = top.winfo_screenheight()
            w = min(720, max(560, sw - 120))
            h = min(520, max(420, sh - 220))
            try:
                rx, ry = self.root.winfo_rootx(), self.root.winfo_rooty()
                rw, rh = self.root.winfo_width(), self.root.winfo_height()
                px = rx + (rw - w) // 2
                py = ry + (rh - h) // 2
            except Exception:
                px = (sw - w) // 2
                py = (sh - h) // 2
            px = max(10, min(px, sw - w - 10))
            py = max(10, min(py, sh - h - 60))
            top.geometry(f"{w}x{h}+{px}+{py}")
            top.minsize(560, 420)
        except Exception:
            top.geometry("640x420")

        outer = ttk.Frame(top, padding=10)
        outer.pack(fill="both", expand=True)

        # Name
        row1 = ttk.Frame(outer)
        row1.pack(fill="x")
        ttk.Label(row1, text="Collector name:", width=14).pack(side="left")
        name_var = tk.StringVar(value=str(initial_name or ""))
        ent_name = ttk.Entry(row1, textvariable=name_var)
        ent_name.pack(side="left", fill="x", expand=True, padx=(6, 0))

        # Areas summary + button
        row2 = ttk.Frame(outer)
        row2.pack(fill="x", pady=(10, 0))
        ttk.Label(row2, text="Route areas:", width=14).pack(side="left")

        areas_state = {"areas": list(initial_areas or [])}
        areas_summary_var = tk.StringVar(value="")

        def _summarize():
            a = areas_state.get("areas") or []
            if not a:
                areas_summary_var.set("0 areas selected")
            elif len(a) <= 4:
                areas_summary_var.set(f"{len(a)} areas: " + ", ".join(a))
            else:
                areas_summary_var.set(f"{len(a)} areas: " + ", ".join(a[:4]) + ", ...")
        _summarize()

        lbl = ttk.Label(row2, textvariable=areas_summary_var, foreground="#555")
        lbl.pack(side="left", fill="x", expand=True, padx=(6, 6))

        def choose_areas():
            picked = self._area_picker_dialog(initial=(areas_state.get("areas") or []), title="Select Route Areas")
            if picked is None:
                return
            areas_state["areas"] = picked
            _summarize()

        ttk.Button(row2, text="Choose / Edit…", command=choose_areas).pack(side="right")

        # Notes
        ttk.Label(outer, text="Notes / Route Instructions:").pack(anchor="w", pady=(12, 4))
        notes = tk.Text(outer, height=10, wrap="word")
        notes.pack(fill="both", expand=True)
        try:
            notes.insert("1.0", str(initial_notes or ""))
        except Exception as __spina_exc:
            _log_suppressed_once('excpass_0237', 'suppressed exception excpass_0237', __spina_exc)
            pass

        # Footer buttons
        footer = ttk.Frame(outer)
        footer.pack(fill="x", pady=(10, 0))

        result = {"ok": False, "name": None, "areas": None, "notes": None}

        def on_ok():
            nm = (name_var.get() or "").strip()
            if not nm:
                messagebox.showwarning("Missing Name", "Please enter a collector name.")
                return
            a = list(areas_state.get("areas") or [])
            if not a:
                messagebox.showwarning("No Areas", "Please select at least one area for this collector.")
                return
            try:
                nt = notes.get("1.0", "end").strip()
            except Exception:
                nt = ""
            result.update({"ok": True, "name": nm, "areas": a, "notes": nt})
            try:
                top.grab_release()
            except Exception as __spina_exc:
                _log_suppressed_once('excpass_0238', 'suppressed exception excpass_0238', __spina_exc)
                pass
            top.destroy()

        def on_cancel():
            result.update({"ok": False})
            try:
                top.grab_release()
            except Exception as __spina_exc:
                _log_suppressed_once('excpass_0239', 'suppressed exception excpass_0239', __spina_exc)
                pass
            top.destroy()

        ttk.Button(footer, text="Cancel", command=on_cancel).pack(side="right", padx=4)
        ttk.Button(footer, text="OK", command=on_ok).pack(side="right")

        top.protocol("WM_DELETE_WINDOW", on_cancel)
        try:
            ent_name.focus_set()
            ent_name.select_range(0, "end")
        except Exception as __spina_exc:
            _log_suppressed_once('excpass_0240', 'suppressed exception excpass_0240', __spina_exc)
            pass

        top.wait_window()
        if result.get("ok"):
            return {"name": result["name"], "areas": result["areas"], "notes": result["notes"]}
        return None
